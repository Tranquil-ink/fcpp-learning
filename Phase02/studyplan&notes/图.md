###  建立无向图
```cpp
int n, m;
cin >> n >> m;

vector<vector<int>> adj(n);

for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;

    adj[u].push_back(v);
    adj[v].push_back(u);
}
```

## 1. BFS：广度优先搜索

```cpp
void bfs(int start) {
    queue<int> q;

    visited[start] = true;
    q.push(start);

    while (!q.empty()) {
        int u = q.front();
        q.pop();

        for (int v : adj[u]) {
            if (!visited[v]) {
                visited[v] = true;
                q.push(v);
            }
        }
    }
}
```
- **起点为什么要先标记？其他点为什么在入队时标记，而不是出队时？**  
一般节点在入队之后不立即出队，若不先标记，前面和他相连节点出队时会将他重复加入队列

## 2. DFS：深度优先搜索
```cpp
void dfs(int u) {
    visited[u] = true;

    for (int v : adj[u]) {
        if (!visited[v]) {
            dfs(v);
        }
    }
}
```
与广度优先遍历类似，深度优先遍历序列的顺序也不是唯一的。给定某顶点，先往哪个方向探索都可以，即邻接顶点的顺序可以任意打乱，都是深度优先遍历。

以树的遍历为例，前序、中序、后序遍历，它们展示了三种遍历优先级，然而这三者都属于深度优先遍历



1. 求图/矩阵中连通块的个数
   - 把矩阵转成图，每个点连接上下左右，用dx[]表示
   - bfs
  
```cpp
#include <iostream>
#include <vector>
#include <queue>
#include <utility>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;

    vector<vector<int>> a(n, vector<int>(m));
    vector<vector<bool>> vis(n, vector<bool>(m, false));

    for (int i = 0; i < n; ++i)
        for (int j = 0; j < m; ++j)
            cin >> a[i][j];

    const int dx[4] = {-1, 1, 0, 0};
    const int dy[4] = {0, 0, -1, 1};

    int answer = 0;

    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) {
            if (a[i][j] != 1 || vis[i][j]) continue;

            ++answer;

            queue<pair<int, int>> q;
            vis[i][j] = true;
            q.push({i, j});

            while (!q.empty()) {
                int x = q.front().first;
                int y = q.front().second;
                q.pop();

                for (int k = 0; k < 4; ++k) {
                    int nx = x + dx[k];
                    int ny = y + dy[k];

                    if (nx < 0 || nx >= n ||
                        ny < 0 || ny >= m)
                        continue;

                    if (a[nx][ny] != 1 || vis[nx][ny])
                        continue;

                    vis[nx][ny] = true;
                    q.push({nx, ny});
                }
            }
        }
    }

    cout << answer << '\n';
}
```
